class Passage(BaseModel):

class AdapterRegistry:
    def __init__(self):
        self._registry: Dict[str, DomainAdapter] = {}

    def register(self, adapter: DomainAdapter) -> None:
        self._registry[adapter.name] = adapter

    def get_adapter(self, domain_name: str) -> DomainAdapter:

# 3. UNIVERSAL FALLBACK WEB SEARCH ENGINE
# ==========================================
class UniversalFallbackScraper:
    @staticmethod
    async def fetch_fallback(query: str, domain_context: str, k: int = 2) -> List[Passage]:
        passages = []
        search_query = f"{query} {domain_context} facts source proof"
        url = f"https://duckduckgo.com{search_query}"
        headers = {"User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"}
        
        async with httpx.AsyncClient(timeout=8.0) as client:
            try:
                resp = await client.get(url, headers=headers)
                if resp.status_code == 200:
                    soup = BeautifulSoup(resp.text, "html.parser")
                    links = soup.find_all("a", class_="result__snippet")
                    titles = soup.find_all("a", class_="result__url")
                    
                    for i in range(min(len(links), k)):
                        snippet_text = links[i].get_text().strip()
                        raw_url = titles[i]["href"] if i < len(titles) else "https://duckduckgo.com"
                        clean_url = raw_url.split("?uddg=")[-1].replace("%3a", ":").replace("%2f", "/")
                        
                        passages.append(Passage(
                            title=f"Web Intelligence Source [{i+1}]",
                            source=f"Live Web Node ({domain_context})",
                            url=clean_url.split("&")[0],
                            publication_date=datetime.datetime.utcnow().strftime("%Y-%m-%d"),
                            snippet=snippet_text
                        ))
            except Exception:
                pass
        return passages

class HealthcareAdapter:
    name = "healthcare"
    async def search(self, query: str, k: int) -> List[Passage]:
        passages = []
        url = "https://nih.gov"
        sum_url = "https://nih.gov"
        
        async with httpx.AsyncClient(timeout=8.0) as client:
            try:
                params = {"db": "pmc", "term": query, "retmode": "json", "retmax": k}
                res = await client.get(url, params=params)
                if res.status_code == 200:
                    id_list = res.json().get("esearchresult", {}).get("idlist", [])
                    if id_list:
                        s_res = await client.get(sum_url, params={"db": "pmc", "id": ",".join(id_list), "retmode": "json"})
                        if s_res.status_code == 200:
                            data = s_res.json().get("result", {})
                            for uid in id_list:
                                passages.append(Passage(
                                    title=data.get(uid, {}).get("title", "Medical Study"),
                                    source="PubMed Central",
                                    url=f"https://nih.gov{uid}/",
                                    publication_date=data.get(uid, {}).get("pubdate", "2026-01-01"),
                                    snippet=data.get(uid, {}).get("summary", "No summary block returned.")
                                ))
            except Exception:
                pass
                
        if not passages:
            return await UniversalFallbackScraper.fetch_fallback(query, "medical clinical research", k)
        return passages[:k]

    def credibility_of(self, source_id: str) -> float:
        return 0.97
